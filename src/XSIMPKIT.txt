// XSIMPKIT 1.7 - exact complex radicals and guarded simplification.
// Paste the entire file into an ordinary PPL program named XSIMPKIT.
// CAS radians / exact / Simplify=None. Entry: XSIMP(expression).
#cas
// XSIMP 1.1 scalar module. HP Prime: expr[1] is the expression head.
// One-based list indices. No dependence on the unavailable sommet command.
// Traversal depth is bounded; individual built-in CAS calls have no time limit.
xshead(xsve):=BEGIN RETURN xsve[1]; END;

// Inspect numeric leaves without evaluating function heads or identifiers.
// State 0 = exact, 1 = a floating leaf, 2 = traversal limit/self operand.
// The string check only skips trees with no possible printed float notation;
// the actual decision is always made by type, not by matching decimal text.
xsapproxstate(xsve,xsvdepth):=BEGIN
  LOCAL xsvt,xsvl,xsvk,xsvs,xsvr,xsvtext;
  xsvt:=type(xsve);
  IF xsvt==DOM_FLOAT THEN RETURN 1; END;
  IF xsvt==DOM_COMPLEX THEN
    IF type(re(xsve))==DOM_FLOAT OR type(im(xsve))==DOM_FLOAT THEN RETURN 1; END;
    RETURN 0;
  END;
  IF xsvt<>DOM_SYMBOLIC AND xsvt<>DOM_LIST THEN RETURN 0; END;
  xsvtext:=string(xsve);
  IF inString(xsvtext,".")==0 AND inString(xsvtext,"E")==0 AND
     inString(xsvtext,"e")==0 AND inString(xsvtext,"inf")==0 AND
     inString(xsvtext,"nan")==0 THEN RETURN 0; END;
  IF xsvdepth>=24 THEN RETURN 2; END;
  IF xsvt==DOM_LIST THEN xsvl:=xsve; ELSE xsvl:=[op(xsve)]; END;
  xsvr:=0;
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])==xsvtext THEN xsvs:=2;
    ELSE xsvs:=xsapproxstate(xsvl[xsvk],xsvdepth+1); END;
    IF xsvs==1 THEN RETURN 1; END;
    IF xsvs==2 THEN xsvr:=2; END;
  END;
  RETURN xsvr;
END;

xsisapprox(xsve):=BEGIN RETURN xsapproxstate(xsve,0)==1; END;

// Reject newly approximate candidates. If a scan is incomplete, preserve the
// original unless it already contains a confirmed floating input.
xsnewapprox(xsve,xsvc):=BEGIN
  IF xsapproxstate(xsvc,0)==0 THEN RETURN 0; END;
  IF xsisapprox(xsve) THEN RETURN 0; END;
  RETURN 1;
END;

// Bit-like states: 1 = a nonreal numeric atom, 2 = a radical, 3 = both.
// Only inspect types/operands; never normalize a complex radical to classify it.
xscomplexradwalk(xsve,xsvdepth):=BEGIN
  LOCAL xsvt,xsvh,xsvl,xsvk,xsvs,xsvimag,xsvrad,xsvparent;
  xsvt:=type(xsve);
  IF xsvt==DOM_COMPLEX OR xsvt==DOM_RAT THEN
    IF string(im(xsve))<>"0" THEN RETURN 1; END;
    RETURN 0;
  END;
  IF xsvt<>DOM_SYMBOLIC AND xsvt<>DOM_LIST THEN RETURN 0; END;
  IF xsvdepth>=24 THEN RETURN 0; END;
  xsvimag:=0; xsvrad:=0;
  IF xsvt==DOM_LIST THEN xsvl:=xsve;
  ELSE
    xsvh:=xshead(xsve); xsvl:=[op(xsve)];
    IF xsvh==sqrt THEN xsvrad:=1; END;
    IF string(xsvh)=="'^'" OR string(xsvh)=="^" THEN
      IF size(xsvl)==2 THEN
        IF type(xsvl[2])==DOM_RAT AND string(im(xsvl[2]))=="0" THEN xsvrad:=1; END;
      END;
    END;
  END;
  xsvparent:=string(xsve);
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])<>xsvparent THEN
      xsvs:=xscomplexradwalk(xsvl[xsvk],xsvdepth+1);
      IF xsvs==1 OR xsvs==3 THEN xsvimag:=1; END;
      IF xsvs==2 OR xsvs==3 THEN xsvrad:=1; END;
      IF xsvimag AND xsvrad THEN RETURN 3; END;
    END;
  END;
  RETURN xsvimag+2*xsvrad;
END;

xscomplexrad(xsve):=BEGIN
  IF type(xsve)<>DOM_SYMBOLIC AND type(xsve)<>DOM_LIST THEN RETURN 0; END;
  IF size(lname(xsve))<>0 THEN RETURN 0; END;
  IF xscomplexradwalk(xsve,0)<>3 THEN RETURN 0; END;
  IF xsisapprox(xsve) THEN RETURN 0; END;
  RETURN 1;
END;

// Preserve the elementary identity sin(u)^2+cos(u)^2=1 on the guarded
// path, without asking simplify to enter a complex algebraic extension.
// This bounded pass pairs only literal squares with the same argument.
xsradtrigsquare(xsve):=BEGIN
  LOCAL xsvh,xsvl,xsvb;
  IF type(xsve)<>DOM_SYMBOLIC THEN RETURN []; END;
  xsvh:=string(xshead(xsve));
  IF xsvh<>"'^'" AND xsvh<>"^" THEN RETURN []; END;
  xsvl:=[op(xsve)];
  IF size(xsvl)<>2 THEN RETURN []; END;
  IF type(xsvl[2])<>DOM_INT OR xsvl[2]<>2 THEN RETURN []; END;
  xsvb:=xsvl[1];
  IF type(xsvb)<>DOM_SYMBOLIC THEN RETURN []; END;
  xsvh:=xshead(xsvb);
  IF xsvh==sin THEN RETURN [1,op(xsvb)]; END;
  IF xsvh==cos THEN RETURN [2,op(xsvb)]; END;
  RETURN [];
END;

xsradtrigunit(xsve):=BEGIN
  LOCAL xsvh,xsvl,xsvk,xsvj,xsva,xsvb,xsvhit,xsvchanged,xsvr;
  IF type(xsve)<>DOM_SYMBOLIC THEN RETURN xsve; END;
  xsvh:=string(xshead(xsve));
  IF xsvh<>"'+'" AND xsvh<>"+" THEN RETURN xsve; END;
  xsvl:=[op(xsve)];
  IF size(xsvl)>32 THEN RETURN xsve; END;
  xsvchanged:=0;
  FOR xsvk FROM 1 TO size(xsvl) DO
    xsva:=xsradtrigsquare(xsvl[xsvk]);
    IF size(xsva)==2 THEN
      xsvhit:=0;
      FOR xsvj FROM xsvk+1 TO size(xsvl) DO
        IF NOT xsvhit THEN
          xsvb:=xsradtrigsquare(xsvl[xsvj]);
          IF size(xsvb)==2 THEN
            IF xsva[1]<>xsvb[1] AND string(xsva[2])==string(xsvb[2]) THEN
              xsvl[xsvk]:=1; xsvl[xsvj]:=0; xsvhit:=1; xsvchanged:=1;
            END;
          END;
        END;
      END;
    END;
  END;
  IF NOT xsvchanged THEN RETURN xsve; END;
  xsvr:=0;
  FOR xsvk FROM 1 TO size(xsvl) DO xsvr:=xsvr+xsvl[xsvk]; END;
  IF xsnewapprox(xsve,xsvr) THEN RETURN xsve; END;
  RETURN xsvr;
END;

xspos(xsve):=BEGIN
  IF type(xsve)==DOM_INT OR type(xsve)==DOM_RAT THEN RETURN xsve>0; END;
  IF size(lname(xsve))!=0 THEN RETURN 0; END;
  IF string(simplify(im(xsve)))!="0" THEN RETURN 0; END;
  RETURN string(simplify(sign(xsve)))=="1";
END;

xslogswalk(xsve,xsvdepth):=BEGIN
  LOCAL xsvl,xsvr,xsvk,xsvj,xsva,xsvparent,xsvh;
  xsvr:=[];
  IF xsvdepth>=24 THEN RETURN xsvr; END;
  IF type(xsve)==DOM_LIST THEN xsvl:=xsve;
  ELSE
    IF type(xsve)<>DOM_SYMBOLIC THEN RETURN xsvr; END;
    xsvh:=xshead(xsve);
    IF type(xsvh)<>DOM_FUNC THEN RETURN xsvr; END;
    IF string(xsvh)=="'of'" OR string(xsvh)=="of" THEN RETURN xsvr; END;
    IF xsvh==ln THEN RETURN [xsve]; END;
    xsvl:=[op(xsve)];
  END;
  xsvparent:=string(xsve);
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])<>xsvparent THEN
      xsva:=xslogswalk(xsvl[xsvk],xsvdepth+1);
      FOR xsvj FROM 1 TO size(xsva) DO
        IF NOT member(xsva[xsvj],xsvr) THEN
          xsvr:=append(xsvr,xsva[xsvj]);
        END;
      END;
    END;
  END;
  RETURN xsvr;
END;

xslogs(xsve):=BEGIN RETURN xslogswalk(xsve,0); END;

xsrwwalk(xsve,xsvdepth):=BEGIN
  LOCAL xsvl,xsvr,xsvk,xsvu,xsvb,xsvc,xsvq,xsvparent,xsvh;
  IF xsvdepth>=24 THEN RETURN xsve; END;
  xsvparent:=string(xsve);
  IF type(xsve)==DOM_LIST THEN
    xsvr:=[];
    FOR xsvk FROM 1 TO size(xsve) DO
      IF string(xsve[xsvk])==xsvparent THEN
        xsvr:=append(xsvr,xsve[xsvk]);
      ELSE
        xsvr:=append(xsvr,xsrwwalk(xsve[xsvk],xsvdepth+1));
      END;
    END;
    RETURN xsvr;
  END;
  IF type(xsve)<>DOM_SYMBOLIC THEN RETURN xsve; END;
  xsvh:=xshead(xsve);
  IF type(xsvh)<>DOM_FUNC THEN RETURN xsve; END;
  // Undefined functions use the internal of head and special argument packing.
  // Leave them intact instead of invoking an internal head with a flat list.
  IF string(xsvh)=="'of'" OR string(xsvh)=="of" THEN RETURN xsve; END;
  xsvl:=[op(xsve)]; xsvr:=[];
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])==xsvparent THEN
      xsvr:=append(xsvr,xsvl[xsvk]);
    ELSE
      xsvr:=append(xsvr,xsrwwalk(xsvl[xsvk],xsvdepth+1));
    END;
  END;
  // Re-applying even an unchanged sqrt head can force an exact complex root
  // to decimals in HP mode. Keep unchanged nodes and guard changed nodes.
  IF string(xsvr)==string(xsvl) THEN xsvq:=xsve;
  ELSE
    xsvq:=xsvh(op(xsvr));
    IF xsnewapprox(xsve,xsvq) THEN RETURN xsve; END;
  END;
  IF type(xsvq)<>DOM_SYMBOLIC THEN RETURN xsvq; END;
  xsvh:=xshead(xsvq);
  IF string(xsvh)=="'^'" OR string(xsvh)=="^" THEN
    xsvl:=[op(xsvq)];
    IF size(xsvl)==2 THEN
      xsvb:=xsvl[1]; xsvc:=xsvl[2];
      IF type(xsvb)<>DOM_LIST AND type(xsvc)<>DOM_LIST THEN
        IF xspos(xsvb) THEN
          IF string(simplify(im(xsvc)))!="0" THEN
            xsvr:=exp(ln(xsvb)*xsvc);
            IF xsnewapprox(xsvq,xsvr) THEN RETURN xsvq; END;
            RETURN xsvr;
          END;
        END;
      END;
    END;
  END;
  IF xsvh<>exp THEN RETURN xsvq; END;
  xsvu:=op(xsvq); xsvl:=xslogs(xsvu);
  FOR xsvk FROM 1 TO size(xsvl) DO
    xsvb:=op(xsvl[xsvk]);
    IF xspos(xsvb) AND xsvb<>1 THEN
      xsvc:=normal(xsvu/xsvl[xsvk]);
      IF NOT xsnewapprox(xsvu,xsvc) AND size(xslogs(xsvc))==0 THEN
        IF string(simplify(im(xsvc)))=="0" THEN
          xsvr:=xsvb^xsvc;
          IF xsnewapprox(xsvq,xsvr) THEN RETURN xsvq; END;
          RETURN xsvr;
        END;
      END;
    END;
  END;
  RETURN xsvq;
END;

xsrewrite(xsve):=BEGIN RETURN xsrwwalk(xsve,0); END;

xsweightwalk(xsve,xsvdepth):=BEGIN
  LOCAL xsvl,xsvk,xsvw,xsvh,xsvparent;
  IF xsvdepth>=24 THEN RETURN 1000; END;
  xsvw:=1;
  IF type(xsve)==DOM_LIST THEN xsvl:=xsve;
  ELSE
    IF type(xsve)<>DOM_SYMBOLIC THEN RETURN 1; END;
    xsvh:=xshead(xsve);
    IF type(xsvh)<>DOM_FUNC THEN RETURN size(string(xsve)); END;
    IF string(xsvh)=="'of'" OR string(xsvh)=="of" THEN RETURN size(string(xsve)); END;
    IF xsvh==re OR xsvh==im OR xsvh==arg THEN xsvw:=12; END;
    IF xsvh==sin OR xsvh==cos THEN xsvw:=3; END;
    xsvl:=[op(xsve)];
  END;
  xsvparent:=string(xsve);
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])==xsvparent THEN
      xsvw:=xsvw+1000;
    ELSE
      xsvw:=xsvw+xsweightwalk(xsvl[xsvk],xsvdepth+1);
    END;
  END;
  RETURN xsvw;
END;

xsweight(xsve):=BEGIN RETURN xsweightwalk(xsve,0); END;
xsscore(xsve):=BEGIN RETURN 20*xsweight(xsve)+size(string(xsve)); END;

// Retain legacy strategy numbering. An empty string means this internal
// candidate repeats step 1 or is inapplicable; xsbounded skips it.
xstrans(xsve,xsvk):=BEGIN
  LOCAL xsvr;
  xsvr:=xsve;
  IFERR
    IF xsvk==1 THEN xsvr:=simplify(xsve); END;
    IF xsvk==2 THEN xsvr:=normal(xsve); END;
    IF xsvk==5 THEN
      xsvr:=trig2exp(xsve);
      IF string(xsvr)==string(xsve) THEN RETURN ""; END;
      xsvr:=simplify(xsvr);
    END;
    IF xsvk==6 THEN
      IF size(lname(xsve))!=0 THEN RETURN ""; END;
      xsvr:=exp2trig(xsve);
      IF string(xsvr)==string(xsve) THEN RETURN ""; END;
      xsvr:=simplify(xsvr);
    END;
    IF xsnewapprox(xsve,xsvr) THEN RETURN xsve; END;
    xsvr:=xsrewrite(xsvr);
    IF type(xsvr)==DOM_STRING THEN RETURN xsve; END;
    IF has(xsvr,undef) OR has(xsvr,infinity) THEN RETURN xsve; END;
  THEN
    RETURN xsve;
  END;
  RETURN xsvr;
END;

// One pass over the same original expression, never a branching search.
xsbounded(xsve,xsvdepth,xsvmatrix):=BEGIN
  LOCAL xsvb,xsvc,xsvk,xsvr,xsvsteps,xsvbest,xsvscore,xsvparent,xsvseen,xsvtext;
  IF xsvdepth>=24 THEN RETURN xsve; END;
  IF type(xsve)==DOM_STRING THEN RETURN xsve; END;
  IF type(xsve)==DOM_LIST THEN
    xsvr:=[]; xsvparent:=string(xsve);
    FOR xsvk FROM 1 TO size(xsve) DO
      IF string(xsve[xsvk])==xsvparent THEN
        xsvr:=append(xsvr,xsve[xsvk]);
      ELSE
        xsvr:=append(xsvr,xsbounded(xsve[xsvk],xsvdepth+1,xsvmatrix));
      END;
    END;
    RETURN xsvr;
  END;
  xsvb:=xsrewrite(xsve);
  IF type(xsvb)<>DOM_SYMBOLIC THEN RETURN xsvb; END;
  // Native normal/simplify can spend seconds in an algebraic extension for
  // sin(s)/s with a closed exact complex radical s, then return decimals.
  // ratnormal treats these function/radical nodes as algebraic atoms.
  // Classify once, before any of the expensive native strategy calls.
  IF xscomplexrad(xsvb) THEN
    xsvb:=xsradtrigunit(xsvb);
    IFERR
      xsvc:=ratnormal(xsvb);
      IF type(xsvc)<>DOM_STRING AND NOT has(xsvc,undef) AND NOT has(xsvc,infinity) AND NOT xsnewapprox(xsvb,xsvc) THEN
        xsvb:=xsradtrigunit(xsvc);
      END;
    THEN RETURN xsvb;
    END;
    RETURN xsvb;
  END;
  xsvtext:=string(xsvb); xsvseen:=[xsvtext];
  IF xsvmatrix THEN
    xsvbest:=size(xsvtext); xsvsteps:=[1,2];
  ELSE
    xsvbest:=20*xsweight(xsvb)+size(xsvtext); xsvsteps:=[1,2,5];
  END;
  IF size(lname(xsve))==0 THEN xsvsteps:=append(xsvsteps,6); END;
  FOR xsvk FROM 1 TO size(xsvsteps) DO
    xsvc:=xstrans(xsve,xsvsteps[xsvk]);
    IF type(xsvc)<>DOM_STRING THEN
      IF type(xsvc)<>DOM_SYMBOLIC AND type(xsvc)<>DOM_LIST THEN RETURN xsvc; END;
      xsvtext:=string(xsvc);
      IF NOT member(xsvtext,xsvseen) THEN
        xsvseen:=append(xsvseen,xsvtext);
        IF xsvmatrix THEN xsvscore:=size(xsvtext);
        ELSE xsvscore:=20*xsweight(xsvc)+size(xsvtext); END;
        IF xsvscore<xsvbest THEN xsvb:=xsvc; xsvbest:=xsvscore; END;
      END;
    END;
  END;
  RETURN xsvb;
END;

xsscalar(xsve):=BEGIN RETURN xsbounded(xsve,0,0); END;
XFAST(xsve):=BEGIN RETURN xsscalar(xsve); END;
// Legacy name retained for replacement installs; it no longer does deep search.
XDEEP(xsve):=BEGIN RETURN xsscalar(xsve); END;
xssearch(xsve,xsvdepth,xsvwidth):=BEGIN RETURN xsscalar(xsve); END;
XREWRITE(xsve):=BEGIN RETURN xsrewrite(xsve); END;
xsclean(xsve):=BEGIN RETURN xsbounded(xsve,0,1); END;

xsrectwalk(xsve,xsvdepth):=BEGIN
  LOCAL xsvk,xsvr,xsvparent;
  IF xsvdepth>=24 THEN RETURN xsve; END;
  IF type(xsve)==DOM_LIST THEN
    xsvr:=[]; xsvparent:=string(xsve);
    FOR xsvk FROM 1 TO size(xsve) DO
      IF string(xsve[xsvk])==xsvparent THEN xsvr:=append(xsvr,xsve[xsvk]);
      ELSE xsvr:=append(xsvr,xsrectwalk(xsve[xsvk],xsvdepth+1)); END;
    END;
    RETURN xsvr;
  END;
  xsvr:=simplify(re(xsve))+i*simplify(im(xsve));
  IF xsnewapprox(xsve,xsvr) THEN RETURN xsve; END;
  RETURN xsvr;
END;

XRECT(xsve):=BEGIN RETURN xsrectwalk(xsve,0); END;


// XSIMP matrix module. CAS angle mode must be radians.
// xsclean/xsisapprox are supplied by scalar; xspnative by presentation.

xsmz(xsmvee):=BEGIN
  LOCAL xsmvs;
  xsmvs:=string(normal(xsmvee));
  RETURN (xsmvs=="0") OR (xsmvs=="0.0");
END;

xsmsign(xsmvee):=BEGIN
  LOCAL xsmvs;
  xsmvs:=string(simplify(sign(xsmvee)));
  IF xsmvs=="1" THEN RETURN 1; END;
  IF xsmvs=="-1" THEN RETURN -1; END;
  RETURN 0;
END;

// Principal square root without native sqrt(Gaussian rational), which can
// silently approximate even in Exact mode. If z=a+i*b and b!=0, set
// rho=sqrt(a^2+b^2), u=sqrt((rho+a)/2), v=sign(b)*sqrt((rho-a)/2).
// Both inner radicands are nonnegative and (u+i*v)^2=z, with u>0.
// Exact real and rational cases keep their native behavior. Floats stay
// approximate; unsupported exact complex constants stay as a held square root.
xsmroot(xsmvz):=BEGIN
  LOCAL xsmva,xsmvb,xsmvrho,xsmvu,xsmvv;
  IF type(xsmvz)==DOM_INT THEN RETURN sqrt(xsmvz); END;
  // Giac can tag (3+2*i)/6 as DOM_RAT: rational storage alone does not
  // establish that the value is real.
  IF type(xsmvz)==DOM_RAT THEN
    IF string(im(xsmvz))=="0" THEN RETURN sqrt(xsmvz); END;
  END;
  IF xsisapprox(xsmvz) THEN RETURN sqrt(xsmvz); END;
  IF size(lname(xsmvz))==0 THEN
    IFERR
      xsmva:=re(xsmvz); xsmvb:=im(xsmvz);
      IF (type(xsmva)==DOM_INT OR type(xsmva)==DOM_RAT) AND
         (type(xsmvb)==DOM_INT OR type(xsmvb)==DOM_RAT) THEN
        IF xsmvb==0 THEN RETURN sqrt(xsmva); END;
        xsmvrho:=sqrt(xsmva^2+xsmvb^2);
        xsmvu:=sqrt(ratnormal((xsmvrho+xsmva)/2));
        xsmvv:=sqrt(ratnormal((xsmvrho-xsmva)/2));
        IF xsmvb<0 THEN xsmvv:=-xsmvv; END;
        RETURN xsmvu+i*xsmvv;
      END;
      // Preserve ordinary real radicals and symbolic real constants.
      IF string(xsmvb)=="0" THEN RETURN sqrt(xsmvz); END;
    THEN
      RETURN expr("quote(sqrt("+xspnative(xsmvz)+"))");
    END;
  END;
  RETURN expr("quote(sqrt("+xspnative(xsmvz)+"))");
END;

xsmzero(xsmva):=BEGIN
  LOCAL xsmvj,xsmvk;
  FOR xsmvj FROM 1 TO rowDim(xsmva) DO
    FOR xsmvk FROM 1 TO colDim(xsmva) DO
      IF NOT xsmz(xsmva[xsmvj,xsmvk]) THEN RETURN 0; END;
    END;
  END;
  RETURN 1;
END;

xsmclean(xsmva):=BEGIN
  LOCAL xsmvj,xsmvk,xsmvr;
  xsmvr:=xsmva;
  FOR xsmvj FROM 1 TO rowDim(xsmva) DO
    FOR xsmvk FROM 1 TO colDim(xsmva) DO
      xsmvr[xsmvj,xsmvk]:=xsclean(xsmva[xsmvj,xsmvk]);
    END;
  END;
  RETURN xsmvr;
END;

// Rewrite only: preserve the Cayley-Hamilton linear-combination layout.
xsmrewrite(xsmva):=BEGIN
  LOCAL xsmvj,xsmvk,xsmvr;
  xsmvr:=xsmva;
  FOR xsmvj FROM 1 TO rowDim(xsmva) DO
    FOR xsmvk FROM 1 TO colDim(xsmva) DO
      xsmvr[xsmvj,xsmvk]:=xsrewrite(xsmva[xsmvj,xsmvk]);
    END;
  END;
  RETURN xsmvr;
END;

xsmid(xsmvn):=BEGIN
  LOCAL xsmvj;
  RETURN diag(seq(1,xsmvj,1,xsmvn));
END;

xsmdiagonal(xsmva):=BEGIN
  LOCAL xsmvj,xsmvk;
  FOR xsmvj FROM 1 TO rowDim(xsmva) DO
    FOR xsmvk FROM 1 TO colDim(xsmva) DO
      IF xsmvj!=xsmvk THEN
        IF NOT xsmz(xsmva[xsmvj,xsmvk]) THEN RETURN 0; END;
      END;
    END;
  END;
  RETURN 1;
END;

// exp(t*A), A=mu*I+B and B^2=r^2*I. Finish coefficient arithmetic
// before attaching hyperbolic functions; the final held output must not
// freeze separate numeric factors such as 2*sinh(u)/sqrt(3)*4.
xsmhyperexp(xsmvt,xsmvmu,xsmvbb,xsmvrr):=BEGIN
  LOCAL xsmvn,xsmvres,xsmvj,xsmvk,xsmvpref,xsmvu,xsmvch,xsmvsh,xsmvc;
  xsmvn:=rowDim(xsmvbb);
  xsmvres:=0*xsmid(xsmvn);
  xsmvpref:=xsclean(exp(normal(xsmvt*xsmvmu)));
  xsmvu:=normal(xsmvt*xsmvrr);
  xsmvch:=cosh(xsmvu); xsmvsh:=sinh(xsmvu);
  FOR xsmvj FROM 1 TO xsmvn DO
    FOR xsmvk FROM 1 TO xsmvn DO
      xsmvc:=xsclean(xsmvbb[xsmvj,xsmvk]/xsmvrr);
      IF xsmvj==xsmvk THEN
        xsmvres[xsmvj,xsmvk]:=xsmvpref*(xsmvch+xsmvc*xsmvsh);
      ELSE
        xsmvc:=xsclean(xsmvpref*xsmvc);
        xsmvres[xsmvj,xsmvk]:=xsmvc*xsmvsh;
      END;
    END;
  END;
  RETURN xsmvres;
END;

// Hyperbolic Cayley-Hamilton form for a positive scalar base and B^2=r^2 I.
// Clean coefficients before introducing cosh/sinh, never the completed entries.
xsmhyperpower(xsmvbas,xsmvmu,xsmvbb,xsmvrr):=BEGIN
  LOCAL xsmvn,xsmvres,xsmvj,xsmvk,xsmvpref,xsmvu,xsmvch,xsmvsh,xsmvc,xsmvrational;
  xsmvn:=rowDim(xsmvbb);
  xsmvres:=0*xsmid(xsmvn);
  xsmvrational:=type(xsmvmu)==DOM_INT OR type(xsmvmu)==DOM_RAT;
  IF xsmvrational THEN xsmvpref:=xsclean(xsmvbas^xsmvmu);
  ELSE xsmvpref:=exp(xsmvmu*ln(xsmvbas)); END;
  xsmvu:=normal(xsmvrr*ln(xsmvbas));
  xsmvch:=cosh(xsmvu); xsmvsh:=sinh(xsmvu);
  FOR xsmvj FROM 1 TO xsmvn DO
    FOR xsmvk FROM 1 TO xsmvn DO
      xsmvc:=xsclean(xsmvbb[xsmvj,xsmvk]/xsmvrr);
      IF xsmvj==xsmvk THEN
        xsmvres[xsmvj,xsmvk]:=xsmvpref*(xsmvch+xsmvc*xsmvsh);
      ELSE
        IF xsmvrational THEN xsmvc:=xsclean(xsmvpref*xsmvc);
        ELSE xsmvc:=xsmvpref*xsmvc; END;
        xsmvres[xsmvj,xsmvk]:=xsmvc*xsmvsh;
      END;
    END;
  END;
  RETURN xsmvres;
END;

// Spectral projector form for B^2=r^2 I. Clean algebraic coefficients only.
// Ordinary scalar powers may immediately become exp on Prime. Keep the two
// exponential spectral values separate until xspresent builds the final held
// power expression; never simplify a rational expression in these values.
xsmpospower(xsmvbas,xsmvmu,xsmvbb,xsmvrr):=BEGIN
  LOCAL xsmvn,xsmvres,xsmvj,xsmvk,xsmvup,xsmvdn,xsmvb,xsmvc,xsmvd,xsmvlup,xsmvldn,xsmvrational,xsmvhyp;
  xsmvn:=rowDim(xsmvbb);
  xsmvres:=0*xsmid(xsmvn);
  xsmvlup:=normal(xsmvmu+xsmvrr); xsmvldn:=normal(xsmvmu-xsmvrr);
  xsmvrational:=1;
  IF type(xsmvlup)==DOM_INT OR type(xsmvlup)==DOM_RAT THEN
    xsmvup:=xsclean(xsmvbas^xsmvlup);
  ELSE
    xsmvrational:=0;
    xsmvup:=exp(xsmvlup*ln(xsmvbas));
  END;
  IF type(xsmvldn)==DOM_INT OR type(xsmvldn)==DOM_RAT THEN
    xsmvdn:=xsclean(xsmvbas^xsmvldn);
  ELSE
    xsmvrational:=0;
    xsmvdn:=exp(xsmvldn*ln(xsmvbas));
  END;
  FOR xsmvj FROM 1 TO xsmvn DO
    FOR xsmvk FROM 1 TO xsmvn DO
      xsmvb:=xsmvbb[xsmvj,xsmvk];
      IF xsmvj==xsmvk THEN
        IF xsmz(xsmvb) THEN
          xsmvres[xsmvj,xsmvk]:=(xsmvup+xsmvdn)/2;
        ELSE
          xsmvc:=xsclean((1+xsmvb/xsmvrr)/2);
          xsmvd:=xsclean((1-xsmvb/xsmvrr)/2);
          xsmvres[xsmvj,xsmvk]:=xsmvc*xsmvup+xsmvd*xsmvdn;
        END;
      ELSE
        xsmvc:=xsclean(xsmvb/(2*xsmvrr));
        xsmvres[xsmvj,xsmvk]:=xsmvc*(xsmvup-xsmvdn);
      END;
    END;
  END;
  IF xsmvrational THEN RETURN xsmclean(xsmvres); END;
  // Compare final text, since ordinary string(exp(...)) overstates the length
  // of a power that the final presentation pass will preserve with quote.
  xsmvhyp:=xsmhyperpower(xsmvbas,xsmvmu,xsmvbb,xsmvrr);
  IF size(xsptext(xsmvhyp,0))<=size(xsptext(xsmvres,0)) THEN RETURN xsmvhyp; END;
  RETURN xsmvres;
END;

// kind=0: exp(t*a); kind=1: bas^a, with positive-real bas.
xsmcore(xsmva,xsmvt,xsmvbas,xsmvkind):=BEGIN
  LOCAL xsmvn,xsmvii,xsmvmu,xsmvbb,xsmvqq,xsmvrr,xsmvcc,xsmvdd,xsmvpref,xsmvj,xsmvk,xsmvpp,xsmvtotal,xsmvup,xsmvdn,xsmvres,xsmvtheta;
  xsmvn:=rowDim(xsmva);
  IF xsmvn!=colDim(xsmva) THEN RETURN "XEXP/XPOW requires a square matrix"; END;
  xsmvii:=xsmid(xsmvn);
  IF xsmdiagonal(xsmva) THEN
    xsmvres:=0*xsmvii;
    FOR xsmvj FROM 1 TO xsmvn DO
      IF xsmvkind==1 THEN xsmvres[xsmvj,xsmvj]:=xsmvbas^xsmva[xsmvj,xsmvj];
      ELSE xsmvres[xsmvj,xsmvj]:=exp(xsmvt*xsmva[xsmvj,xsmvj]); END;
    END;
    RETURN xsmclean(xsmvres);
  END;
  xsmvmu:=xsclean(trace(xsmva)/xsmvn);
  xsmvbb:=xsmva-xsmvmu*xsmvii;
  xsmvqq:=xsclean((xsmvbb*xsmvbb)[1,1]);
  IF xsmvkind==1 THEN xsmvpref:=xsmvbas^xsmvmu;
  ELSE xsmvpref:=exp(xsmvt*xsmvmu); END;
  IF xsmzero(xsmvbb*xsmvbb-xsmvqq*xsmvii) THEN
    IF xsmz(xsmvqq) THEN RETURN xsmclean(xsmvpref*(xsmvii+xsmvt*xsmvbb)); END;
    IF size(lname(xsmvqq))!=0 THEN
      RETURN xsmclean(exp(xsmvt*xsmva));
    END;
    IF xsmsign(xsmvqq)==-1 THEN
      xsmvrr:=sqrt(-xsmvqq);
      xsmvcc:=cos(xsmvt*xsmvrr);
      xsmvdd:=sin(xsmvt*xsmvrr)/xsmvrr;
      RETURN xsmclean(xsmvpref*(xsmvcc*xsmvii+xsmvdd*xsmvbb));
    END;
    IF xsmsign(xsmvqq)==1 THEN
      xsmvrr:=sqrt(xsmvqq);
      IF xsmvkind==1 THEN
        RETURN xsmpospower(xsmvbas,xsmvmu,xsmvbb,xsmvrr);
      END;
      // exp(i*u*A), A with B^2=qI and q>0: prefer ordinary trig.
      // Decide before generating cosh(i*u), which Prime may leave unchanged.
      IF xsmz(re(xsmvt)) THEN
        xsmvtheta:=normal(-i*xsmvt*xsmvrr);
        RETURN xsmclean(xsmvpref*(cos(xsmvtheta)*xsmvii+i*sin(xsmvtheta)/xsmvrr*xsmvbb));
      END;
      RETURN xsmhyperexp(xsmvt,xsmvmu,xsmvbb,xsmvrr);
    END;
    // A nonzero complex constant q: scalar trig form is still valid.
    xsmvrr:=xsmroot(-xsmvqq);
    xsmvcc:=cos(xsmvt*xsmvrr);
    xsmvdd:=sin(xsmvt*xsmvrr)/xsmvrr;
    RETURN xsmclean(xsmvpref*(xsmvcc*xsmvii+xsmvdd*xsmvbb));
  END;
  // Shifted nilpotent matrices: exact terminating Taylor polynomial.
  xsmvpp:=xsmvii; xsmvtotal:=xsmvii;
  FOR xsmvj FROM 1 TO xsmvn DO
    xsmvpp:=xsmvpp*xsmvbb;
    IF xsmzero(xsmvpp) THEN RETURN xsmclean(xsmvpref*xsmvtotal); END;
    xsmvtotal:=xsmvtotal+xsmvt^xsmvj/xsmvj!*xsmvpp;
  END;
  // General higher order: invoke the genuine matrix exponential.
  RETURN xsmclean(exp(xsmvt*xsmva));
END;

XEXP(xsmva):=BEGIN
  RETURN xsmcore(xsmva,1,0,0);
END;

XEXPT(xsmva,xsmvt):=BEGIN
  RETURN xsmcore(xsmva,xsmvt,0,0);
END;

XPOW(xsmvbas,xsmva):=BEGIN
  IF size(lname(xsmvbas))!=0 THEN
    RETURN "XPOW requires a positive real constant base";
  END;
  IF NOT xsmz(im(xsmvbas)) THEN
    RETURN "XPOW requires a positive real base";
  END;
  IF xsmsign(xsmvbas)!=1 THEN
    RETURN "XPOW requires a provably positive real base";
  END;
  RETURN xsmcore(xsmva,ln(xsmvbas),xsmvbas,1);
END;

// Genuine matrix sine/cosine, never elementwise sin(a) or cos(a).
xsmtrigcore(xsmva,xsmvkind):=BEGIN
  LOCAL xsmvn,xsmvii,xsmvmu,xsmvbb,xsmvqq,xsmvrr,xsmvcc,xsmvdd,xsmvj,xsmvpp,xsmvtotal,xsmvsm,xsmvcm,xsmvres;
  xsmvn:=rowDim(xsmva);
  IF xsmvn!=colDim(xsmva) THEN RETURN "XSIN/XCOS requires a square matrix"; END;
  xsmvii:=xsmid(xsmvn);
  IF xsmdiagonal(xsmva) THEN
    xsmvres:=0*xsmvii;
    FOR xsmvj FROM 1 TO xsmvn DO
      IF xsmvkind==0 THEN xsmvres[xsmvj,xsmvj]:=sin(xsmva[xsmvj,xsmvj]);
      ELSE xsmvres[xsmvj,xsmvj]:=cos(xsmva[xsmvj,xsmvj]); END;
    END;
    RETURN xsmclean(xsmvres);
  END;
  xsmvmu:=xsclean(trace(xsmva)/xsmvn);
  xsmvbb:=xsmva-xsmvmu*xsmvii;
  xsmvqq:=xsclean((xsmvbb*xsmvbb)[1,1]);
  xsmvsm:=sin(xsmvmu); xsmvcm:=cos(xsmvmu);
  IF xsmzero(xsmvbb*xsmvbb-xsmvqq*xsmvii) THEN
    IF xsmz(xsmvqq) THEN
      IF xsmvkind==0 THEN RETURN xsmclean(xsmvsm*xsmvii+xsmvcm*xsmvbb); END;
      RETURN xsmclean(xsmvcm*xsmvii-xsmvsm*xsmvbb);
    END;
    IF size(lname(xsmvqq))==0 THEN
      IF xsmsign(xsmvqq)==-1 THEN
        xsmvrr:=sqrt(-xsmvqq); xsmvcc:=cosh(xsmvrr); xsmvdd:=sinh(xsmvrr)/xsmvrr;
      ELSE
        xsmvrr:=xsmroot(xsmvqq); xsmvcc:=cos(xsmvrr); xsmvdd:=sin(xsmvrr)/xsmvrr;
      END;
      IF xsmvkind==0 THEN RETURN xsmclean(xsmvsm*xsmvcc*xsmvii+xsmvcm*xsmvdd*xsmvbb); END;
      RETURN xsmclean(xsmvcm*xsmvcc*xsmvii-xsmvsm*xsmvdd*xsmvbb);
    END;
  END;
  // Exact Taylor series for a scalar plus a nilpotent matrix.
  xsmvpp:=xsmvii;
  IF xsmvkind==0 THEN xsmvtotal:=xsmvsm*xsmvii; ELSE xsmvtotal:=xsmvcm*xsmvii; END;
  FOR xsmvj FROM 1 TO xsmvn DO
    xsmvpp:=xsmvpp*xsmvbb;
    IF xsmzero(xsmvpp) THEN RETURN xsmclean(xsmvtotal); END;
    IF xsmvkind==0 THEN xsmvtotal:=xsmvtotal+sin(xsmvmu+xsmvj*pi/2)/xsmvj!*xsmvpp;
    ELSE xsmvtotal:=xsmvtotal+cos(xsmvmu+xsmvj*pi/2)/xsmvj!*xsmvpp; END;
  END;
  IF xsmvkind==0 THEN RETURN xsmclean((exp(i*xsmva)-exp(-i*xsmva))/(2*i)); END;
  RETURN xsmclean((exp(i*xsmva)+exp(-i*xsmva))/2);
END;

XSIN(xsmva):=BEGIN
  RETURN xsmtrigcore(xsmva,0);
END;

XCOS(xsmva):=BEGIN
  RETURN xsmtrigcore(xsmva,1);
END;


// Final presentation pass. Build text first, then parse once under quote.
// No algebraic simplification may follow xspresent: native ^ evaluation may
// otherwise turn a real irrational power straight back into exp(c*ln(b)).
// Bounds apply to traversal/text; individual native CAS calls have no timeout.

// HP-mode string(i) can be a private-use display glyph that expr cannot parse.
// Translate that glyph outside string literals. Keep other native syntax.
xspnative(xspv):=BEGIN
  LOCAL xsps,xspr,xspk,xspn,xspi,xspni,xspch,xspquoted,xspescape;
  xsps:=string(xspv);
  IF type(xspv)==DOM_STRING THEN
    xsps:=string([xspv]); xsps:=mid(xsps,2,size(xsps)-2);
    // The printed string is already quoted/escaped. Its contents are data,
    // so no imaginary-symbol translation can be needed inside this literal.
    RETURN xsps;
  END;
  xspi:=string(i); xspni:=size(xspi);
  IF xspi=="i" THEN RETURN xsps; END;
  IF size(xsps)>6400 THEN RETURN xsps; END;
  // Prime's inString is one-based and returns 0 when absent. Most expression
  // nodes contain no private imaginary glyph: avoid a CAS-level character loop.
  IF inString(xsps,xspi)==0 THEN RETURN xsps; END;
  xspr:=""; xspk:=1; xspn:=size(xsps); xspquoted:=0; xspescape:=0;
  WHILE xspk<=xspn DO
    xspch:=mid(xsps,xspk,1);
    IF xspquoted==0 AND mid(xsps,xspk,xspni)==xspi THEN
      xspr:=xspr+"(i)"; xspk:=xspk+xspni;
    ELSE
      xspr:=xspr+xspch; xspk:=xspk+1;
      IF xspescape==1 THEN xspescape:=0;
      ELSE
        IF xspch=="\\" THEN xspescape:=1;
        ELSE IF xspch=="\"" THEN xspquoted:=1-xspquoted; END; END;
      END;
    END;
  END;
  RETURN xspr;
END;

// Only functions with ordinary unary call syntax are reconstructed here.
// Internal/special operators retain their native printed syntax unchanged.
xspunaryname(xsph):=BEGIN
  IF xsph==exp THEN RETURN "exp"; END;
  IF xsph==ln THEN RETURN "ln"; END;
  IF xsph==sqrt THEN RETURN "sqrt"; END;
  IF xsph==sin THEN RETURN "sin"; END;
  IF xsph==cos THEN RETURN "cos"; END;
  IF xsph==tan THEN RETURN "tan"; END;
  IF xsph==asin THEN RETURN "asin"; END;
  IF xsph==acos THEN RETURN "acos"; END;
  IF xsph==atan THEN RETURN "atan"; END;
  IF xsph==sinh THEN RETURN "sinh"; END;
  IF xsph==cosh THEN RETURN "cosh"; END;
  IF xsph==tanh THEN RETURN "tanh"; END;
  IF xsph==asinh THEN RETURN "asinh"; END;
  IF xsph==acosh THEN RETURN "acosh"; END;
  IF xsph==atanh THEN RETURN "atanh"; END;
  IF xsph==abs THEN RETURN "abs"; END;
  IF xsph==re THEN RETURN "re"; END;
  IF xsph==im THEN RETURN "im"; END;
  IF xsph==arg THEN RETURN "arg"; END;
  IF xsph==conj THEN RETURN "conj"; END;
  IF xsph==sign THEN RETURN "sign"; END;
  RETURN "";
END;

// Find a shorter trigonometric argument using identities valid for every
// complex u: cosh(u)=cos(-i*u), sinh(u)=i*sin(-i*u). Never evaluate re(u):
// some native/upstream builds fail on re(sqrt(3)*ln(a)) with complex variables.
// The leading space makes the absence test work in both zero-/one-based modes.
xsptrigarg(xspu):=BEGIN
  LOCAL xsps,xspc,xspneg,xspsign,xspnew;
  xsps:=string(xspu);
  IF size(xsps)>256 THEN RETURN []; END;
  IF inString(" "+xsps,string(i))<=0 THEN RETURN []; END;
  xspc:=-i*xspu;
  // Even normal(-u) can approximate an exact complex radical on Prime.
  // A shorter display argument must retain the input's exactness.
  IF xsnewapprox(xspu,xspc) THEN RETURN []; END;
  xspnew:=normal(xspc);
  IF NOT xsnewapprox(xspc,xspnew) THEN xspc:=xspnew; END;
  xspneg:=-xspc; xspnew:=normal(xspneg); xspsign:=1;
  IF NOT xsnewapprox(xspneg,xspnew) THEN xspneg:=xspnew; END;
  IF NOT xsnewapprox(xspc,xspneg) AND size(string(xspneg))<size(string(xspc)) THEN xspc:=xspneg; xspsign:=-1; END;
  IF size(string(xspc))<size(xsps) THEN RETURN [xspc,xspsign]; END;
  RETURN [];
END;

// Convert small hyperbolic leaves before rendering their parents. Odd/even
// trig identities need no sign assumptions. Only changed arithmetic parents
// are regrouped; unchanged hyperbolic expressions are left intact.
xspimagwalk(xspv,xspd):=BEGIN
  LOCAL xsph,xspname,xspl,xspr,xspk,xspq,xspu,xspc,xspneg,xspchanged,xspnew,xspparent;
  xspparent:=string(xspv);
  IF xspd>=24 OR size(xspparent)>1600 THEN RETURN xspv; END;
  IF type(xspv)==DOM_LIST THEN
    xspr:=xspv;
    FOR xspk FROM 1 TO size(xspv) DO
      IF string(xspv[xspk])==xspparent THEN RETURN xspv; END;
      xspr[xspk]:=xspimagwalk(xspv[xspk],xspd+1);
    END;
    RETURN xspr;
  END;
  IF type(xspv)<>DOM_SYMBOLIC THEN RETURN xspv; END;
  xsph:=xshead(xspv);
  IF type(xsph)<>DOM_FUNC THEN RETURN xspv; END;
  xspname:=string(xsph);
  IF xspname=="'of'" OR xspname=="of" THEN RETURN xspv; END;
  IF xspunaryname(xsph)=="" AND xspname<>"'+'" AND xspname<>"+" AND xspname<>"'-'" AND xspname<>"-" AND xspname<>"'*'" AND xspname<>"*" AND xspname<>"'/'" AND xspname<>"/" AND xspname<>"'inv'" AND xspname<>"inv" AND xspname<>"'^'" AND xspname<>"^" THEN RETURN xspv; END;
  xspl:=[op(xspv)]; xspr:=[]; xspchanged:=0;
  // Do not evaluate an unevaluated matrix function/product in a display pass.
  FOR xspk FROM 1 TO size(xspl) DO
    IF type(xspl[xspk])==DOM_LIST THEN RETURN xspv; END;
    IF string(xspl[xspk])==xspparent THEN RETURN xspv; END;
  END;
  FOR xspk FROM 1 TO size(xspl) DO
    xspnew:=xspimagwalk(xspl[xspk],xspd+1);
    IF string(xspnew)<>string(xspl[xspk]) THEN xspchanged:=1; END;
    xspr:=append(xspr,xspnew);
  END;
  xspq:=xspv;
  IF xspchanged THEN
    xspnew:=xsph(op(xspr));
    IF NOT xsnewapprox(xspv,xspnew) THEN xspq:=xspnew; END;
  END;
  IF size(xspr)==1 AND size(string(xspr[1]))<=256 THEN
    xspu:=xspr[1];
    IF xsph==cosh OR xsph==sinh THEN
      xspc:=xsptrigarg(xspu);
      IF size(xspc)==2 THEN
        IF xsph==cosh THEN xspnew:=cos(xspc[1]);
        ELSE xspnew:=xspc[2]*i*sin(xspc[1]); END;
        IF NOT xsnewapprox(xspv,xspnew) THEN RETURN xspnew; END;
      END;
    END;
    IF xsph==sin OR xsph==cos THEN
      xspneg:=-xspu; xspnew:=normal(xspneg);
      IF NOT xsnewapprox(xspneg,xspnew) THEN xspneg:=xspnew; END;
      IF NOT xsnewapprox(xspu,xspneg) AND size(string(xspneg))<size(string(xspu)) THEN
        IF xsph==cos THEN xspnew:=cos(xspneg);
        ELSE xspnew:=-sin(xspneg); END;
        IF NOT xsnewapprox(xspv,xspnew) THEN RETURN xspnew; END;
      END;
    END;
  END;
  IF xspchanged THEN
    IF xspname=="'+'" OR xspname=="+" OR xspname=="'-'" OR xspname=="-" OR xspname=="'*'" OR xspname=="*" OR xspname=="'/'" OR xspname=="/" OR xspname=="'inv'" OR xspname=="inv" THEN
      xspnew:=regroup(regroup(xspq));
      IF type(xspnew)<>DOM_STRING AND NOT has(xspnew,undef) AND NOT has(xspnew,infinity) AND NOT xsnewapprox(xspv,xspnew) THEN RETURN xspnew; END;
    END;
  END;
  RETURN xspq;
END;

// Held parsing can retain nested multiplication nodes. Flatten only those
// nodes, keeping factor order; no expansion or algebraic evaluation is used.
xspflatproduct(xspv,xspd):=BEGIN
  LOCAL xsph,xspl,xspr,xspk,xspj,xspsub;
  IF xspd>=24 OR type(xspv)<>DOM_SYMBOLIC THEN RETURN [xspv]; END;
  xsph:=string(xshead(xspv));
  IF xsph<>"'*'" AND xsph<>"*" THEN RETURN [xspv]; END;
  xspl:=[op(xspv)]; xspr:=[];
  FOR xspk FROM 1 TO size(xspl) DO
    IF string(xspl[xspk])==string(xspv) THEN RETURN [xspv]; END;
    xspsub:=xspflatproduct(xspl[xspk],xspd+1);
    FOR xspj FROM 1 TO size(xspsub) DO
      xspr:=append(xspr,xspsub[xspj]);
      IF size(xspr)>256 THEN RETURN [xspv]; END;
    END;
  END;
  RETURN xspr;
END;

// Render scalar products as one fraction without evaluating the product.
// Matrix products/inverses retain their original order and formatting path.
xspproducttext(xspv,xspl,xspd):=BEGIN
  LOCAL xspk,xspf,xsph,xspa,xspnum,xspden,xspnf,xspdf,xspnt,xspdt;
  xspl:=xspflatproduct(xspv,xspd);
  xspnum:=""; xspden:="";
  FOR xspk FROM 1 TO size(xspl) DO
    xspf:=xspl[xspk];
    IF type(xspf)==DOM_LIST THEN RETURN ""; END;
    IF string(xspf)==string(xspv) THEN RETURN ""; END;
    xspnf:=xspf; xspdf:=1;
    IF type(xspf)==DOM_RAT THEN
      xspnf:=numer(xspf); xspdf:=denom(xspf);
    ELSE
      IF type(xspf)==DOM_SYMBOLIC THEN
        xsph:=string(xshead(xspf));
        IF xsph=="'inv'" OR xsph=="inv" THEN
          xspdf:=op(xspf);
          IF type(xspdf)==DOM_LIST THEN RETURN ""; END;
          xspnf:=1;
        ELSE
          IF xsph=="'^'" OR xsph=="^" OR xsph=="'/'" OR xsph=="/" THEN
            xspa:=[op(xspf)];
            IF size(xspa)==2 THEN
              IF type(xspa[1])==DOM_LIST OR type(xspa[2])==DOM_LIST THEN RETURN ""; END;
              IF xsph=="'/'" OR xsph=="/" THEN
                xspnf:=xspa[1]; xspdf:=xspa[2];
              ELSE
                IF string(xspa[2])=="-1" THEN xspnf:=1; xspdf:=xspa[1]; END;
              END;
            END;
          END;
        END;
      END;
    END;
    IF string(xspnf)<>"1" THEN
      xspnt:=xsptext(xspnf,xspd+1);
      IF xspnum<>"" THEN xspnum:=xspnum+"*"; END;
      xspnum:=xspnum+"("+xspnt+")";
    END;
    IF string(xspdf)<>"1" THEN
      xspdt:=xsptext(xspdf,xspd+1);
      IF xspden<>"" THEN xspden:=xspden+"*"; END;
      xspden:=xspden+"("+xspdt+")";
    END;
    IF size(xspnum)+size(xspden)>6400 THEN RETURN ""; END;
  END;
  IF xspnum=="" THEN xspnum:="1"; END;
  IF xspden=="" THEN RETURN "("+xspnum+")"; END;
  RETURN "("+xspnum+")/("+xspden+")";
END;

xsptext(xspv,xspd):=BEGIN
  LOCAL xsps,xsph,xspname,xspl,xspr,xspk,xspu,xspb,xspc,xspn,xspopen,xspclose,xspchild,xspreal;
  xsps:=xspnative(xspv);
  IF xspd>=24 OR size(xsps)>1600 THEN RETURN xsps; END;
  IF type(xspv)==DOM_LIST THEN
    xspopen:=mid(xsps,1,1);
    IF xspopen=="[" THEN xspclose:="]";
    ELSE IF xspopen=="{" THEN xspclose:="}";
    ELSE RETURN xsps; END; END;
    xspr:=xspopen;
    FOR xspk FROM 1 TO size(xspv) DO
      IF xspk>1 THEN xspr:=xspr+","; END;
      IF string(xspv[xspk])==string(xspv) THEN RETURN xsps; END;
      xspr:=xspr+xsptext(xspv[xspk],xspd+1);
      IF size(xspr)>6400 THEN RETURN xsps; END;
    END;
    RETURN xspr+xspclose;
  END;
  IF type(xspv)<>DOM_SYMBOLIC THEN RETURN xsps; END;
  xsph:=xshead(xspv);
  IF type(xsph)<>DOM_FUNC THEN RETURN xsps; END;
  xspname:=xspunaryname(xsph);
  IF xspname<>"" THEN
    xspu:=op(xspv);
    // A unary matrix operand is a sequence of rows in Prime's op output.
    IF type(xspu)==DOM_LIST THEN xspu:=[xspu]; END;
    IF string(xspu)==string(xspv) THEN RETURN xsps; END;
    IF type(xspu)<>DOM_LIST THEN
      IF xsph==exp THEN
        xspl:=xslogs(xspu);
        IF size(xspl)<=8 THEN
          FOR xspk FROM 1 TO size(xspl) DO
            xspb:=op(xspl[xspk]);
            // This same logarithm already occurs in the input. Re-express its
            // scalar multiple as a principal power, retaining the whole base.
            // ln(base) excludes base=0; do not split products or nested powers.
            IF type(xspb)<>DOM_LIST AND xspb<>0 AND xspb<>1 AND NOT has(xspb,undef) AND NOT has(xspb,infinity) THEN
              xspc:=ratnormal(xspu/xspl[xspk]);
              IF NOT xsnewapprox(xspu,xspc) AND size(xslogs(xspc))==0 THEN
                xspchild:=im(xspc); xspreal:=simplify(xspchild);
                IF NOT xsnewapprox(xspc,xspchild) AND NOT xsnewapprox(xspchild,xspreal) THEN
                  IF string(xspreal)=="0" THEN
                    RETURN "("+xsptext(xspb,xspd+1)+")^("+xsptext(xspc,xspd+1)+")";
                  END;
                END;
              END;
            END;
          END;
        END;
      END;
      IF xsph==cosh OR xsph==sinh THEN
        xspc:=xsptrigarg(xspu);
        IF size(xspc)==2 THEN
          IF xsph==cosh THEN RETURN "cos("+xsptext(xspc[1],xspd+1)+")"; END;
          IF xspc[2]==-1 THEN RETURN "(-i)*sin("+xsptext(xspc[1],xspd+1)+")"; END;
          RETURN "(i)*sin("+xsptext(xspc[1],xspd+1)+")";
        END;
      END;
    END;
    RETURN xspname+"("+xsptext(xspu,xspd+1)+")";
  END;
  xspname:=string(xsph);
  IF mid(xspname,1,1)=="'" THEN xspname:=mid(xspname,2,size(xspname)-2); END;
  IF xspname<>"+" AND xspname<>"-" AND xspname<>"*" AND xspname<>"/" AND xspname<>"^" AND xspname<>"=" AND xspname<>"==" AND xspname<>"<" AND xspname<>">" AND xspname<>"<=" AND xspname<>">=" AND xspname<>"!=" AND xspname<>"<>" THEN RETURN xsps; END;
  xspl:=[op(xspv)]; xspn:=size(xspl);
  IF xspn==0 THEN RETURN xsps; END;
  IF xspname<>"+" AND xspname<>"*" AND xspname<>"-" AND xspn<>2 THEN RETURN xsps; END;
  IF xspname=="-" AND xspn>2 THEN RETURN xsps; END;
  IF xspname=="*" THEN
    xspr:=xspproducttext(xspv,xspl,xspd);
    IF xspr<>"" THEN RETURN xspr; END;
  END;
  // normal may store sqrt(c) as c^(1/2); retain the familiar radical spelling.
  IF xspname=="^" AND xspn==2 THEN
    IF type(xspl[1])<>DOM_LIST AND string(xspl[2])=="1/2" THEN
      RETURN "sqrt("+xsptext(xspl[1],xspd+1)+")";
    END;
  END;
  xspr:="";
  IF xspname=="-" AND xspn==1 THEN xspr:="-"; END;
  FOR xspk FROM 1 TO xspn DO
    IF string(xspl[xspk])==string(xspv) THEN RETURN xsps; END;
    IF xspk>1 THEN xspr:=xspr+xspname; END;
    xspr:=xspr+"("+xsptext(xspl[xspk],xspd+1)+")";
    IF size(xspr)>6400 THEN RETURN xsps; END;
  END;
  RETURN "("+xspr+")";
END;

// Isolated parser shim: no evaluation/normal/simplify after this call.
xsphold(xsps):=BEGIN RETURN expr("quote("+xsps+")"); END;

xspresent(xspv):=BEGIN
  LOCAL xsps,xspclean,xspresult;
  IF type(xspv)==DOM_STRING THEN RETURN xspv; END;
  IFERR
    xspclean:=xspimagwalk(xspv,0);
    IF xsnewapprox(xspv,xspclean) THEN xspclean:=xspv; END;
    xsps:=xsptext(xspclean,0);
    IF size(xsps)>6400 THEN RETURN xspv; END;
    xspresult:=xsphold(xsps);
    IF xsnewapprox(xspv,xspresult) THEN RETURN xspv; END;
    RETURN xspresult;
  THEN
    RETURN xspv;
  END;
END;


// Recover compact hyperbolic or trigonometric forms in evaluated matrices.
// Only a Laurent polynomial a/z+b+c*z, z=exp(u), is accepted.
// No matrix logarithm, floating recognition, or recovery of input intent.
xsppa(xsve,xsvdepth):=BEGIN
  LOCAL xsvh,xsvl,xsvr,xsvk,xsvj,xsvpart;
  xsvr:=[];
  IF xsvdepth>=24 THEN RETURN [0,0,0]; END;
  IF type(xsve)!=DOM_SYMBOLIC THEN RETURN xsvr; END;
  xsvh:=xsve[1];
  IF xsvh==exp THEN RETURN [xsve]; END;
  IF type(xsvh)!=DOM_FUNC THEN RETURN [0,0,0]; END;
  IF string(xsvh)=="of" OR string(xsvh)=="'of'" THEN RETURN [0,0,0]; END;
  xsvl:=[op(xsve)];
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])==string(xsve) THEN RETURN [0,0,0]; END;
    xsvpart:=xsppa(xsvl[xsvk],xsvdepth+1);
    IF size(xsvpart)>2 THEN RETURN [0,0,0]; END;
    FOR xsvj FROM 1 TO size(xsvpart) DO
      IF NOT member(xsvpart[xsvj],xsvr) THEN xsvr:=append(xsvr,xsvpart[xsvj]); END;
      IF size(xsvr)>2 THEN RETURN [0,0,0]; END;
    END;
  END;
  RETURN xsvr;
END;

// pow2exp may introduce ln(base). For non-rational exponents, admit only
// positive real constant bases; do not add branches for an unknown base.
xspppowers(xsve,xsvdepth):=BEGIN
  LOCAL xsvh,xsvl,xsvk;
  IF xsvdepth>=24 THEN RETURN 0; END;
  IF type(xsve)!=DOM_SYMBOLIC THEN RETURN 1; END;
  xsvh:=string(xsve[1]); xsvl:=[op(xsve)];
  IF xsvh=="'^'" OR xsvh=="^" THEN
    IF size(xsvl)!=2 THEN RETURN 0; END;
    IF type(xsvl[2])!=DOM_INT AND type(xsvl[2])!=DOM_RAT THEN
      IF NOT xspos(xsvl[1]) THEN RETURN 0; END;
    END;
  END;
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])==string(xsve) THEN RETURN 0; END;
    IF NOT xspppowers(xsvl[xsvk],xsvdepth+1) THEN RETURN 0; END;
  END;
  RETURN 1;
END;

// Denominators may only contain products/powers of nonzero constants and
// exponentials. A difference of exponentials might secretly be zero: decline.
xsppnz(xsve,xsvdepth):=BEGIN
  LOCAL xsvh,xsvl,xsvk,xsvs;
  IF xsvdepth>=24 THEN RETURN 0; END;
  IF size(xsppa(xsve,0))==0 THEN
    IF size(lname(xsve))!=0 THEN RETURN 0; END;
    xsvs:=string(normal(sign(xsve)));
    RETURN xsvs=="1" OR xsvs=="-1";
  END;
  IF type(xsve)!=DOM_SYMBOLIC THEN RETURN 0; END;
  xsvh:=xsve[1];
  IF xsvh==exp THEN RETURN 1; END;
  xsvl:=[op(xsve)];
  IF string(xsvh)=="'^'" OR string(xsvh)=="^" THEN
    IF size(xsvl)==2 AND type(xsvl[2])==DOM_INT THEN RETURN xsppnz(xsvl[1],xsvdepth+1); END;
  END;
  IF string(xsvh)!="'*'" AND string(xsvh)!="*" THEN RETURN 0; END;
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF NOT xsppnz(xsvl[xsvk],xsvdepth+1) THEN RETURN 0; END;
  END;
  RETURN 1;
END;

xsppsafe(xsve,xsvdepth):=BEGIN
  LOCAL xsvh,xsvl,xsvk;
  IF xsvdepth>=24 THEN RETURN 0; END;
  IF type(xsve)!=DOM_SYMBOLIC THEN RETURN 1; END;
  xsvh:=string(xsve[1]); xsvl:=[op(xsve)];
  IF xsvh=="'inv'" OR xsvh=="inv" THEN
    IF NOT xsppnz(xsvl[1],0) THEN RETURN 0; END;
  END;
  IF xsvh=="'/'" OR xsvh=="/" THEN
    IF size(xsvl)!=2 THEN RETURN 0; END;
    IF NOT xsppnz(xsvl[2],0) THEN RETURN 0; END;
  END;
  IF xsvh=="'^'" OR xsvh=="^" THEN
    IF size(xsvl)!=2 THEN RETURN 0; END;
    IF type(xsvl[2])==DOM_INT OR type(xsvl[2])==DOM_RAT THEN
      IF xsvl[2]<0 THEN
        IF NOT xsppnz(xsvl[1],0) THEN RETURN 0; END;
      END;
    END;
  END;
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])==string(xsve) THEN RETURN 0; END;
    IF NOT xsppsafe(xsvl[xsvk],xsvdepth+1) THEN RETURN 0; END;
  END;
  RETURN 1;
END;

// A cheap structural hint for the optional pow2exp fallback. Integer and
// rational powers alone do not justify another CAS transformation pass.
xspppowhint(xsve,xsvdepth):=BEGIN
  LOCAL xsvh,xsvl,xsvk;
  IF xsvdepth>=24 OR type(xsve)!=DOM_SYMBOLIC THEN RETURN 0; END;
  xsvh:=xsve[1];
  IF type(xsvh)!=DOM_FUNC THEN RETURN 0; END;
  IF string(xsvh)=="of" OR string(xsvh)=="'of'" THEN RETURN 0; END;
  xsvl:=[op(xsve)];
  IF string(xsvh)=="'^'" OR string(xsvh)=="^" THEN
    IF size(xsvl)==2 THEN
      IF type(xsvl[2])!=DOM_INT AND type(xsvl[2])!=DOM_RAT THEN RETURN 1; END;
    END;
  END;
  FOR xsvk FROM 1 TO size(xsvl) DO
    IF string(xsvl[xsvk])==string(xsve) THEN RETURN 0; END;
    IF xspppowhint(xsvl[xsvk],xsvdepth+1) THEN RETURN 1; END;
  END;
  RETURN 0;
END;

// Recover from the supplied exponential atoms without changing logarithms.
// The identities exp(m+u)=exp(m)*exp(u) and cosh/sinh definitions hold for
// finite complex arguments, including arguments containing an existing ln(a).
xspprecover(xsve,xsvf,xsva,xsvlen):=BEGIN
  LOCAL xsvu,xsvv,xsvm,xsvp,xsvz,xsvpoly,xsvden,xsv0,xsv1,xsv2,xsvc,xsvs,xsvr,xsvk,xsvtrig,xsvangle,xsvrlen,xsvtlen;
  IF size(xsva)<1 OR size(xsva)>2 THEN RETURN xsve; END;
  IFERR
    // Do not assign or purge a global. Decline if the scratch identifier is
    // already assigned or occurs in the input expression.
    xsvz:=expr("quote(xsppostauxv)");
    IF type(eval(xsvz))!=DOM_IDENT OR string(eval(xsvz))!="xsppostauxv" THEN RETURN xsve; END;
    IF has(xsve,xsvz) THEN RETURN xsve; END;
    IF NOT xsppsafe(xsvf,0) THEN RETURN xsve; END;
    xsvu:=op(xsva[1]);
    IF size(xsva)==1 THEN
      xsvf:=subst(xsvf,xsva[1],xsvz);
    ELSE
      xsvv:=op(xsva[2]);
      xsvm:=normal((xsvu+xsvv)/2);
      xsvu:=normal((xsvu-xsvv)/2);
      xsvp:=xsclean(exp(xsvm));
      xsvf:=subst(xsvf,xsva,[xsvp*xsvz,xsvp/xsvz]);
    END;
    IF size(xsppa(xsvf,0))>0 THEN
      // A nonconstant center exp(m) is allowed, but substitution must remove
      // both original atoms. The polynomial proof below rejects residual z.
      FOR xsvk FROM 1 TO size(xsva) DO
        IF has(xsvf,xsva[xsvk]) THEN RETURN xsve; END;
      END;
    END;
    xsvpoly:=ratnormal(xsvf*xsvz);
    IF size(string(xsvpoly))>3000 THEN RETURN xsve; END;
    xsvden:=denom(xsvpoly);
    IF has(xsvden,xsvz) THEN RETURN xsve; END;
    xsv0:=coeff(xsvpoly,xsvz,0);
    xsv1:=coeff(xsvpoly,xsvz,1);
    xsv2:=coeff(xsvpoly,xsvz,2);
    IF has(xsv0,xsvz) OR has(xsv1,xsvz) OR has(xsv2,xsvz) THEN RETURN xsve; END;
    IF string(ratnormal(xsvpoly-xsv0-xsv1*xsvz-xsv2*xsvz^2))!="0" THEN RETURN xsve; END;
    xsvc:=normal(xsv0+xsv2); xsvs:=normal(xsv2-xsv0);
    xsvr:=normal(xsv1)+xsvc*cosh(xsvu)+xsvs*sinh(xsvu);
    // cosh(u)=cos(-i*u), sinh(u)=i*sin(-i*u), for every complex u.
    // Clean only the angle and coefficients, never the completed candidate.
    xsvangle:=normal(-i*xsvu);
    xsvtrig:=normal(xsv1)+xsvc*cos(xsvangle)+normal(i*xsvs)*sin(xsvangle);
    xsvrlen:=size(string(xsvr)); xsvtlen:=size(string(xsvtrig));
    // A short decimal is not a successful simplification of an exact input.
    IF xsnewapprox(xsve,xsvr) THEN xsvr:=xsve; xsvrlen:=xsvlen; END;
    IF xsnewapprox(xsve,xsvtrig) THEN xsvtrig:=xsve; xsvtlen:=xsvlen; END;
    IF xsvtlen<xsvrlen THEN xsvr:=xsvtrig; xsvrlen:=xsvtlen; END;
    IF has(xsvr,undef) OR has(xsvr,infinity) THEN RETURN xsve; END;
    IF xsvrlen<xsvlen THEN RETURN xsvr; END;
  THEN RETURN xsve;
  END;
  RETURN xsve;
END;

xsppostscalar(xsve):=BEGIN
  LOCAL xsvf,xsva,xsvr,xsvoldlogs,xsvnewlogs,xsvk,xsvtext,xsvlen;
  IF type(xsve)!=DOM_SYMBOLIC THEN RETURN xsve; END;
  xsvtext:=string(xsve); xsvlen:=size(xsvtext);
  IF xsvlen>1000 THEN RETURN xsve; END;
  IF has(xsve,undef) OR has(xsve,infinity) THEN RETURN xsve; END;
  IFERR
    // First use the original expression. An already present logarithm is
    // part of its domain; it does not require a positive constant argument.
    xsva:=xsppa(xsve,0);
    IF size(xsva)>=1 AND size(xsva)<=2 THEN
      xsvr:=xspprecover(xsve,xsve,xsva,xsvlen);
      IF string(xsvr)!=xsvtext THEN RETURN xsvr; END;
    END;
    IF NOT xspppowhint(xsve,0) THEN RETURN xsve; END;
    IF NOT xspppowers(xsve,0) THEN RETURN xsve; END;
    xsvoldlogs:=xslogs(xsve);
    xsvf:=pow2exp(xsve);
    IF xsnewapprox(xsve,xsvf) THEN RETURN xsve; END;
    IF string(xsvf)==xsvtext OR size(string(xsvf))>2000 THEN RETURN xsve; END;
    xsvnewlogs:=xslogs(xsvf);
    IF size(xsvnewlogs)>8 THEN RETURN xsve; END;
    FOR xsvk FROM 1 TO size(xsvnewlogs) DO
      IF NOT member(xsvnewlogs[xsvk],xsvoldlogs) THEN
        IF NOT xspos(op(xsvnewlogs[xsvk])) THEN RETURN xsve; END;
      END;
    END;
    xsva:=xsppa(xsvf,0);
    RETURN xspprecover(xsve,xsvf,xsva,xsvlen);
  THEN RETURN xsve;
  END;
END;

xsmpost(xsmv):=BEGIN
  LOCAL xsmvr,xsmvi,xsmvj;
  IF NOT xsumatrix(xsmv) THEN RETURN xsmv; END;
  IF size(string(xsmv))>1600 THEN RETURN xsmv; END;
  xsmvr:=xsmv;
  FOR xsmvi FROM 1 TO rowDim(xsmv) DO
    FOR xsmvj FROM 1 TO colDim(xsmv) DO
      xsmvr[xsmvi,xsmvj]:=xsppostscalar(xsmv[xsmvi,xsmvj]);
    END;
  END;
  RETURN xsmvr;
END;


// Unified held-expression dispatcher. HP Prime uses one-based indexing.
// Supply XSIMP("2^a") or XSIMP('2^a') to intercept matrix functions before CAS evaluation.
xsumatrix(xsuinput):=BEGIN
  LOCAL xsuj,xsun;
  IF type(xsuinput)!=DOM_LIST THEN RETURN 0; END;
  IF size(xsuinput)==0 THEN RETURN 0; END;
  IF type(xsuinput[1])!=DOM_LIST THEN RETURN 0; END;
  xsun:=size(xsuinput[1]);
  IF xsun==0 THEN RETURN 0; END;
  FOR xsuj FROM 1 TO size(xsuinput) DO
    IF type(xsuinput[xsuj])!=DOM_LIST THEN RETURN 0; END;
    IF size(xsuinput[xsuj])!=xsun THEN RETURN 0; END;
  END;
  RETURN 1;
END;

xsutree0(xsuinput,xsulevel):=BEGIN
  LOCAL xsuhead,xsulist,xsures,xsuj,xsuarg,xsupow,xsucoef,xsumat,xsufound,xsufact;
  IF xsulevel>24 THEN RETURN "XSIMP: expression depth exceeds 24"; END;
  IF size(string(xsuinput))>1600 THEN RETURN "XSIMP: expression exceeds 1600 characters"; END;
  IF type(xsuinput)==DOM_STRING THEN RETURN xsuinput; END;
  IF type(xsuinput)==DOM_LIST THEN
    xsures:=[];
    FOR xsuj FROM 1 TO size(xsuinput) DO
      // Recover recognized exponential pairs before scalar normalization can
      // expand them again or invoke algebraic-extension simplification.
      xsuarg:=xsuinput[xsuj]; xsupow:=xsuarg;
      IF type(xsuarg)!=DOM_LIST THEN xsupow:=xsppostscalar(xsuarg); END;
      IF string(xsupow)!=string(xsuarg) THEN
        xsuarg:=xsupow;
      ELSE
        xsuarg:=xsutree0(xsuarg,xsulevel+1);
        IF type(xsuarg)==DOM_STRING THEN RETURN xsuarg; END;
        IF type(xsuarg)!=DOM_LIST THEN
          xsuarg:=xsscalar(xsuarg);
          // Scalar cleanup can expose exponential pairs hidden in sin/cos.
          xsuarg:=xsppostscalar(xsuarg);
        END;
      END;
      xsures:=append(xsures,xsuarg);
    END;
    RETURN xsures;
  END;
  IF type(xsuinput)==DOM_IDENT THEN
    xsuarg:=eval(xsuinput);
    IF xsumatrix(xsuarg) THEN RETURN xsutree0(xsuarg,xsulevel+1); END;
    RETURN xsuarg;
  END;
  IF type(xsuinput)!=DOM_SYMBOLIC THEN RETURN xsuinput; END;
  xsuhead:=xsuinput[1];
  IF type(xsuhead)!=DOM_FUNC THEN RETURN "XSIMP: unsupported expression operator"; END;
  IF string(xsuhead)=="of" OR string(xsuhead)=="'of'" THEN RETURN "XSIMP: user function calls are not supported in quoted input"; END;
  // Pull scalar factors out before evaluating exp(t*A), preserving CH structure.
  IF xsuhead==exp THEN
    xsuarg:=op(xsuinput);
    IF type(xsuarg)==DOM_SYMBOLIC THEN
      IF string(xsuarg[1])=="'*'" OR string(xsuarg[1])=="*" THEN
        xsufact:=[op(xsuarg)]; xsucoef:=1; xsufound:=0;
        FOR xsuj FROM 1 TO size(xsufact) DO
          xsupow:=xsutree0(xsufact[xsuj],xsulevel+1);
          IF type(xsupow)==DOM_STRING THEN RETURN xsupow; END;
          IF xsumatrix(xsupow) THEN
            IF xsufound==0 THEN xsumat:=xsupow; xsufound:=1;
            ELSE xsumat:=xsumat*xsupow; END;
          ELSE xsucoef:=xsucoef*xsupow; END;
        END;
        IF xsufound==1 THEN RETURN XEXPT(xsumat,xsucoef); END;
      END;
    END;
  END;
  // Unary matrix arguments must not be flattened into an argument sequence.
  IF xsuhead==exp OR xsuhead==sin OR xsuhead==cos OR xsuhead==sinh OR xsuhead==cosh OR xsuhead==sqrt THEN
    IF string(op(xsuinput))==string(xsuinput) THEN RETURN "XSIMP: expression traversal did not progress"; END;
    xsuarg:=op(xsuinput);
    IF type(xsuarg)==DOM_LIST THEN xsuarg:=[xsuarg]; END;
    xsuarg:=xsutree0(xsuarg,xsulevel+1);
    IF type(xsuarg)==DOM_STRING THEN RETURN xsuarg; END;
    IF xsumatrix(xsuarg) THEN
      IF xsuhead==exp THEN RETURN XEXP(xsuarg); END;
      IF xsuhead==sin THEN RETURN XSIN(xsuarg); END;
      IF xsuhead==cos THEN RETURN XCOS(xsuarg); END;
      IF xsuhead==sinh THEN RETURN (XEXP(xsuarg)-XEXP(-xsuarg))/2; END;
      IF xsuhead==cosh THEN RETURN (XEXP(xsuarg)+XEXP(-xsuarg))/2; END;
    END;
    IF xsuhead==sqrt AND type(xsuarg)!=DOM_LIST THEN RETURN xsmroot(xsuarg); END;
    xsupow:=xsuhead(xsuarg);
    IF xsnewapprox(xsuinput,xsupow) THEN RETURN xsuinput; END;
    RETURN xsupow;
  END;
  xsulist:=[op(xsuinput)]; xsures:=[];
  FOR xsuj FROM 1 TO size(xsulist) DO
    IF string(xsulist[xsuj])==string(xsuinput) THEN RETURN "XSIMP: expression traversal did not progress"; END;
    xsuarg:=xsutree0(xsulist[xsuj],xsulevel+1);
    IF type(xsuarg)==DOM_STRING THEN RETURN xsuarg; END;
    xsures:=append(xsures,xsuarg);
  END;
  IF string(xsuhead)=="'^'" OR string(xsuhead)=="^" THEN
    IF size(xsures)==2 THEN
      xsuarg:=xsures[1]; xsupow:=xsures[2];
      IF xsumatrix(xsupow) THEN
        IF NOT xsumatrix(xsuarg) THEN RETURN XPOW(xsuarg,xsupow); END;
      END;
      IF type(xsupow)==DOM_RAT AND xsupow==1/2 AND type(xsuarg)!=DOM_LIST THEN RETURN xsmroot(xsuarg); END;
    END;
  END;
  xsuarg:=xsuhead(op(xsures));
  IF xsnewapprox(xsuinput,xsuarg) THEN RETURN xsuinput; END;
  RETURN xsuarg;
END;

// Prime expression-tree compatibility checks. No simplify/factor/recursive walk here.
XSCHECK(xsuunused):=BEGIN
  LOCAL xsut,xsuh,xsua;
  xsut:=expr("quote(sin(xsprobevar))");
  IF type(xsut)!=DOM_SYMBOLIC THEN RETURN "XSIMP: quoted expression parsing failed"; END;
  xsuh:=xsut[1];
  IF type(xsuh)!=DOM_FUNC THEN RETURN "XSIMP: root operator extraction failed"; END;
  IF xsuh!=sin THEN RETURN "XSIMP: root operator is not sin"; END;
  xsua:=[op(xsut)];
  IF size(xsua)!=1 THEN RETURN "XSIMP: operand extraction failed"; END;
  IF string(xsua[1])!="xsprobevar" THEN RETURN "XSIMP: operand indexing failed"; END;
  RETURN "XSIMP 1.7 CHECK OK";
END;

xsutree(xsuinput):=BEGIN
  IF type(xsuinput)==DOM_STRING THEN
    RETURN xsutree0(expr("quote("+xsuinput+")"),0);
  END;
  RETURN xsutree0(xsuinput,0);
END;

// Collect obvious sums/products after matrix composition, before holding the
// final display. regroup is deliberately less invasive than normal/simplify.
xsufold(xsue,xsud):=BEGIN
  LOCAL xsur,xsuk,xsuc;
  IF xsud>=24 THEN RETURN xsue; END;
  IF type(xsue)==DOM_LIST THEN
    xsur:=xsue;
    FOR xsuk FROM 1 TO size(xsue) DO
      xsur[xsuk]:=xsufold(xsue[xsuk],xsud+1);
    END;
    RETURN xsur;
  END;
  IF type(xsue)!=DOM_SYMBOLIC OR size(string(xsue))>1600 THEN RETURN xsue; END;
  IFERR
    // A first pass may turn (c+c)/2 into (1/2)*2*c on Prime;
    // the second bounded pass finishes that newly exposed multiplication.
    xsuc:=regroup(regroup(xsue));
    IF type(xsuc)!=DOM_STRING AND NOT has(xsuc,undef) AND NOT has(xsuc,infinity) AND NOT xsnewapprox(xsue,xsuc) THEN
      // Candidate selection must not run the full presentation pass twice.
      IF size(string(xsuc))<size(string(xsue)) THEN RETURN xsuc; END;
    END;
  THEN RETURN xsue;
  END;
  RETURN xsue;
END;

XSIMP(xsuinput):=BEGIN
  LOCAL xsures,xsucheck;
  xsucheck:=XSCHECK(0);
  IF xsucheck!="XSIMP 1.7 CHECK OK" THEN RETURN xsucheck; END;
  IF size(string(xsuinput))>1600 THEN RETURN "XSIMP: input exceeds 1600 characters"; END;
  xsures:=xsutree(xsuinput);
  IF type(xsures)==DOM_STRING THEN RETURN xsures; END;
  // xsutree already simplified every leaf of a list or matrix. Do not run
  // the complete scalar strategy set again over non-matrix lists.
  IF type(xsures)==DOM_LIST THEN RETURN xspresent(xsufold(xsures,0)); END;
  RETURN xspresent(xsufold(xsscalar(xsures),0));
END;

// Compact one-call report for the user's native CAS. No matrix exponentials.
XSDIAG(xsuunused):=BEGIN
  RETURN ["1.7",XSCHECK(0),xsptext(exp((2-sqrt(3))*ln(2)),0),string(xspresent(exp((2-sqrt(3))*ln(2))))];
END;
